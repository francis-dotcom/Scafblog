import requests
import json
from datetime import datetime

class LLMSentimentAnalyzer:
    def __init__(self, model="llama3.2:3b"):
        self.model = model
        self.ollama_url = "http://localhost:11434/api/generate"

    def analyze_news(self, headlines):
        """
        Analyze list of news headlines for market sentiment.

        Args:
            headlines: List of strings

        Returns:
            dict with sentiment score, reasoning, and confidence
        """
        prompt = f"""You are a financial sentiment analyzer for Indian stock markets.

Analyze these news headlines and provide:
1. Overall sentiment: BULLISH / BEARISH / NEUTRAL
2. Sentiment score: 1-10 (10 = extremely bullish)
3. Key reasoning in 2 sentences
4. Confidence: HIGH / MEDIUM / LOW

Headlines:
{chr(10).join(f"- {h}" for h in headlines)}

Respond in JSON format:
{{
  "sentiment": "BULLISH/BEARISH/NEUTRAL",
  "score": 1-10,
  "reasoning": "2 sentences",
  "confidence": "HIGH/MEDIUM/LOW"
}}"""

        payload = {
            "model": self.model,
            "prompt": prompt,
            "stream": False,
            "options": {
                "temperature": 0.1,  # Low temp = consistent output
                "num_predict": 200   # Short response
            }
        }

        try:
            response = requests.post(
                self.ollama_url,
                json=payload,
                timeout=30
            )
            result = json.loads(response.text)
            return json.loads(result['response'])
        except Exception as e:
            print(f"LLM analysis failed: {e}")
            return {
                "sentiment": "NEUTRAL",
                "score": 5,
                "reasoning": "LLM analysis unavailable",
                "confidence": "LOW"
            }

    def analyze_with_fallback(self, headlines):
        """
        Run LLM analysis with keyword fallback.
        If LLM fails, use simple keyword matching.
        """
        # Try LLM first
        result = self.analyze_news(headlines)

        if result['confidence'] == 'LOW':
            # Fallback to keywords
            bullish_words = ['surge', 'rally', 'gain', 'up', 'rise', 'positive', 'growth', 'profit']
            bearish_words = ['fall', 'drop', 'crash', 'down', 'loss', 'negative', 'decline', 'fear']

            bullish_count = sum(1 for h in headlines for w in bullish_words if w in h.lower())
            bearish_count = sum(1 for h in headlines for w in bearish_words if w in h.lower())

            if bullish_count > bearish_count:
                result = {"sentiment": "BULLISH", "score": 6, "reasoning": "Keyword-based fallback", "confidence": "LOW"}
            elif bearish_count > bullish_count:
                result = {"sentiment": "BEARISH", "score": 4, "reasoning": "Keyword-based fallback", "confidence": "LOW"}

        return result

class HybridSignalGenerator:
    def __init__(self):
        self.sentiment_analyzer = LLMSentimentAnalyzer()
        self.technical_weight = 0.7  # Technical = 70% weight
        self.sentiment_weight = 0.3  # Sentiment = 30% weight

    def generate_signal(self, technical_signal, headlines):
        """
        Combine technical and sentiment signals.

        Args:
            technical_signal: dict with 'signal' (BUY/SELL/HOLD) and 'confidence' (0-1)
            headlines: List of news headlines

        Returns:
            Final signal with confidence
        """
        sentiment = self.sentiment_analyzer.analyze_with_fallback(headlines)

        # Convert sentiment to score
        sentiment_score = sentiment['score'] / 10  # 0-1

        # Convert technical signal to score
        tech_score = technical_signal['confidence']
        if technical_signal['signal'] == 'SELL':
            tech_score *= -1

        # Weighted combination
        final_score = (tech_score * self.technical_weight) + (sentiment_score * self.sentiment_weight)

        # Generate final signal
        if final_score > 0.6:
            final_signal = "STRONG BUY"
        elif final_score > 0.3:
            final_signal = "BUY"
        elif final_score < -0.6:
            final_signal = "STRONG SELL"
        elif final_score < -0.3:
            final_signal = "SELL"
        else:
            final_signal = "HOLD"

        return {
            'signal': final_signal,
            'score': round(final_score, 2),
            'technical': technical_signal['signal'],
            'sentiment': sentiment['sentiment'],
            'reasoning': sentiment['reasoning']
        }

import requests
from bs4 import BeautifulSoup

class NSENewsScraper:
    def __init__(self):
        self.base_url = "https://www.nseindia.com"
        self.headers = {
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
        }

    def get_market_news(self, limit=20):
        """Get latest market news from NSE"""
        url = f"{self.base_url}/api/news"
        try:
            response = requests.get(url, headers=self.headers, timeout=10)
            data = response.json()
            headlines = [item['headline'] for item in data['data'][:limit]]
            return headlines
        except Exception as e:
            print(f"NSE news fetch failed: {e}")
            # Fallback: Moneycontrol scraper
            return self._moneycontrol_fallback(limit)

    def _moneycontrol_fallback(self, limit=20):
        """Fallback news source"""
        url = "https://www.moneycontrol.com/markets/indian-indices/"
        response = requests.get(url, headers=self.headers, timeout=10)
        soup = BeautifulSoup(response.text, 'lxml')
        headlines = []
        for item in soup.select('.gutter_box .grey_text')[:limit]:
            headlines.append(item.text.strip())
        return headlines

### The Need for Local AI Solutions in Trading

The integration of Artificial Intelligence (AI) into stock trading has revolutionized the landscape, yet retail traders in India continue to face significant challenges. My personal trading journey revealed a critical gap in the availability of AI-driven tools tailored specifically for the Indian market. Most existing solutions are either cloud-based—introducing latency and privacy concerns—or prohibitively expensive, as seen with platforms like Bloomberg Terminal. This realization led to the development of the nse_ai_agent, an AI trading assistant that utilizes local Large Language Models (LLMs) for real-time sentiment analysis, thus enhancing trading decisions while maintaining user control.

### Understanding Local LLMs: Advantages Over Cloud-Based Solutions

Local LLMs, such as Llama 3.2 and Phi-3 Mini, present several advantages over their cloud-based counterparts. The most significant benefit is the elimination of latency; local models operate directly on the device, avoiding the delays associated with API calls, which can range from 500 to 2000 milliseconds—critical in trading scenarios where timing is paramount. Furthermore, local LLMs incur no costs for API usage, a considerable advantage for retail traders who often work within tight budgets. Privacy is another crucial aspect; local models ensure that sensitive trading data remains on the device, thereby reducing risks associated with data breaches. Additionally, local models are free from rate limits, allowing traders to analyze extensive datasets without constraints, making them ideal for high-frequency trading scenarios. This autonomy empowers traders to make decisions in real-time without reliance on external servers.

### Introducing the nse_ai_agent: Features and Functionalities

The nse_ai_agent was meticulously designed to address these challenges by amalgamating multiple functionalities into a single, user-friendly application. At its core, the agent employs local LLMs to conduct sentiment analysis on the latest news headlines impacting the Indian stock market. The workflow initiates with a cron job that triggers at market open, fetching real-time data and news. The LLM processes this information to generate sentiment scores, which are subsequently integrated with traditional technical indicators such as momentum and the Relative Strength Index (RSI). This hybrid approach empowers traders to make informed decisions based on a comprehensive analysis of both market sentiment and technical data. For example, if a recent news article highlights positive earnings reports from a major company, the sentiment score generated could indicate bullish sentiment, which can then be cross-referenced with technical indicators for a more robust trading signal.
